cmake_minimum_required(VERSION 3.16)
project(waypoint_mission)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(mavros_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(OpenCV REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
add_executable(waypoint_mission src/waypoint_mission.cpp src/gcu_control.c)
ament_target_dependencies(waypoint_mission rclcpp ament_index_cpp geometry_msgs nav_msgs std_msgs mavros_msgs tf2)
target_link_libraries(waypoint_mission yaml-cpp Threads::Threads)
add_executable(waypoint_photo src/waypoint_photo.cpp)
ament_target_dependencies(waypoint_photo rclcpp sensor_msgs std_msgs cv_bridge)
target_link_libraries(waypoint_photo ${OpenCV_LIBRARIES})
install(TARGETS waypoint_mission waypoint_photo DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY missions launch DESTINATION share/${PROJECT_NAME})
ament_package()
