cmake_minimum_required(VERSION 3.8)
project(aruco_test)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc calib3d aruco)
find_package(Threads REQUIRED)
add_executable(Safina_aruco_test src/Safina_aruco_test.cpp)
target_include_directories(Safina_aruco_test PRIVATE include ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(Safina_aruco_test rclcpp sensor_msgs std_msgs geometry_msgs cv_bridge tf2 tf2_geometry_msgs)
target_link_libraries(Safina_aruco_test ${OpenCV_LIBRARIES} Threads::Threads)
install(TARGETS Safina_aruco_test DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
