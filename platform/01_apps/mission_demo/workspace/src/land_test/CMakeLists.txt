cmake_minimum_required(VERSION 3.8)
project(land_test)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(mavros_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2 REQUIRED)
find_package(Threads REQUIRED)
add_executable(Safina_land_test src/Safina_land_test.cpp include/gcu/gcu_control.c)
target_include_directories(Safina_land_test PRIVATE include)
ament_target_dependencies(Safina_land_test rclcpp geometry_msgs nav_msgs mavros_msgs std_msgs std_srvs tf2)
target_link_libraries(Safina_land_test Threads::Threads)
install(TARGETS Safina_land_test DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  foreach(name landing_task mode_start_gate motion_limits takeoff_replay)
    add_executable(${name}_test test/${name}_test.cpp)
    target_include_directories(${name}_test PRIVATE include)
    if(name STREQUAL "takeoff_replay")
      add_test(NAME ${name} COMMAND ${name}_test ${CMAKE_CURRENT_SOURCE_DIR}/test/data/takeoff_hover_20261027.csv)
    else()
      add_test(NAME ${name} COMMAND ${name}_test)
    endif()
  endforeach()
endif()
ament_package()
