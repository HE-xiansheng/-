cmake_minimum_required(VERSION 3.8)
project(rtsp_test)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil)
find_path(MPP_INCLUDE_DIR rk_mpi.h PATH_SUFFIXES rockchip mpp REQUIRED)
find_library(MPP_LIBRARY NAMES rockchip_mpp mpp REQUIRED)
add_executable(rtsp_hw_pub_ultra src/rtsp_hw_pub_ultra.cpp)
target_include_directories(rtsp_hw_pub_ultra PRIVATE ${MPP_INCLUDE_DIR} ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(rtsp_hw_pub_ultra rclcpp sensor_msgs std_msgs cv_bridge)
target_link_libraries(rtsp_hw_pub_ultra ${OpenCV_LIBRARIES} PkgConfig::FFMPEG ${MPP_LIBRARY})
install(TARGETS rtsp_hw_pub_ultra DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(frame_rate_gate_test test/frame_rate_gate_test.cpp)
  target_include_directories(frame_rate_gate_test PRIVATE src)
  add_test(NAME frame_rate_gate_jitter COMMAND frame_rate_gate_test)
endif()
ament_package()
