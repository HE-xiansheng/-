cmake_minimum_required(VERSION 3.8)
project(yolo_ultra_test)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(mavros_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(OpenCV REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/TrackedTarget.msg" DEPENDENCIES std_msgs)
add_library(rknn_runtime SHARED IMPORTED)
set_target_properties(rknn_runtime PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/librknn_api/aarch64/librknnrt.so")
add_library(yoloUltra_lib SHARED src/engine/rknn_engine.cpp src/process/preprocess.cpp src/process/postprocess.cpp src/yolo_det_task/yoloUltra_custom.cpp)
target_include_directories(yoloUltra_lib PUBLIC src librknn_api/include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(yoloUltra_lib rknn_runtime ${OpenCV_LIBRARIES})
add_executable(Safina_yoloUltra_test src/Safina_yoloUltra_test.cpp)
target_include_directories(Safina_yoloUltra_test PRIVATE include)
ament_target_dependencies(Safina_yoloUltra_test rclcpp sensor_msgs std_msgs mavros_msgs cv_bridge)
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(Safina_yoloUltra_test yoloUltra_lib "${cpp_typesupport_target}")
set_target_properties(yoloUltra_lib Safina_yoloUltra_test PROPERTIES INSTALL_RPATH "$ORIGIN:$ORIGIN/private")
install(TARGETS yoloUltra_lib Safina_yoloUltra_test LIBRARY DESTINATION lib/${PROJECT_NAME} RUNTIME DESTINATION lib/${PROJECT_NAME})
install(FILES librknn_api/aarch64/librknnrt.so DESTINATION lib/${PROJECT_NAME}/private)
install(DIRECTORY config weights launch DESTINATION share/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()
