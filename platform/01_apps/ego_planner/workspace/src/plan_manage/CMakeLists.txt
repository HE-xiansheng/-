cmake_minimum_required(VERSION 3.8)
project(plan_manage)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(traj_utils REQUIRED)
find_package(traj_opt REQUIRED)
find_package(plan_env REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(mavros_msgs REQUIRED)
find_package(quadrotor_msgs REQUIRED)
add_library(ego_planner_manager SHARED src/planner_manager.cpp)
target_include_directories(ego_planner_manager PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> ${EIGEN3_INCLUDE_DIRS})
ament_target_dependencies(ego_planner_manager rclcpp traj_utils traj_opt plan_env)
install(TARGETS ego_planner_manager EXPORT export_ego_planner_manager LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_ego_planner_manager HAS_LIBRARY_TARGET)
ament_export_dependencies(rclcpp traj_utils traj_opt plan_env Eigen3)
add_executable(planner_node src/planner_node.cpp src/replan_fsm.cpp)
target_link_libraries(planner_node ego_planner_manager)
ament_target_dependencies(planner_node mavros_msgs quadrotor_msgs)
install(TARGETS planner_node RUNTIME DESTINATION lib/${PROJECT_NAME})
ament_package()
