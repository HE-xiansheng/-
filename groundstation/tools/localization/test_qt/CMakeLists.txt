cmake_minimum_required(VERSION 3.21)
project(localization_controls_test LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick)
get_filename_component(GCS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
qt_add_executable(check_localization check.cpp "${GCS_ROOT}/localizationcontroller.cpp" "${GCS_ROOT}/localizationcontroller.h")
target_include_directories(check_localization PRIVATE "${GCS_ROOT}")
target_compile_definitions(check_localization PRIVATE GCS_QML_DIR="${GCS_ROOT}")
target_link_libraries(check_localization PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick)
qt_add_resources(check_localization helper PREFIX /qt/qml/CustomGeometryExample BASE "${GCS_ROOT}" FILES "${GCS_ROOT}/tools/localization/manage_stack.py")
