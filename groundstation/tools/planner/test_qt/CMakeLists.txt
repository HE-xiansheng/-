cmake_minimum_required(VERSION 3.21)
project(planner_ui_test LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml)
get_filename_component(GCS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
qt_add_executable(check_planner_ui check.cpp "${GCS_ROOT}/plannercontroller.cpp" "${GCS_ROOT}/plannercontroller.h")
target_include_directories(check_planner_ui PRIVATE "${GCS_ROOT}")
target_link_libraries(check_planner_ui PRIVATE Qt6::Core Qt6::Gui Qt6::Qml)
qt_add_resources(check_planner_ui guard PREFIX /qt/qml/CustomGeometryExample BASE "${GCS_ROOT}" FILES "${GCS_ROOT}/tools/planner/stop_preview.py")
