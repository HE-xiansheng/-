cmake_minimum_required(VERSION 3.21)
project(preflight_link_test LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Network)
get_filename_component(GCS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
qt_add_executable(check_preflight_link check.cpp "${GCS_ROOT}/connectioncontroller.cpp" "${GCS_ROOT}/connectioncontroller.h" "${GCS_ROOT}/preflightreceiver.cpp" "${GCS_ROOT}/preflightreceiver.h")
target_include_directories(check_preflight_link PRIVATE "${GCS_ROOT}")
target_compile_definitions(check_preflight_link PRIVATE GCS_SOURCE_DIR="${GCS_ROOT}")
target_link_libraries(check_preflight_link PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Network)
