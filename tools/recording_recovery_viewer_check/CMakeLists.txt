cmake_minimum_required(VERSION 3.21)
project(recording_recovery_viewer_check LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Quick Quick3D Network Concurrent)
find_package(SQLite3 REQUIRED)
get_filename_component(GCS "${CMAKE_CURRENT_SOURCE_DIR}/../../groundstation" ABSOLUTE)
add_executable(check_recovered main.cpp ${GCS}/pointcloudfileloader.cpp ${GCS}/pointcloudfileloader.h ${GCS}/pointcloudgeometry.cpp ${GCS}/pointcloudgeometry.h)
target_include_directories(check_recovered PRIVATE ${GCS})
target_compile_definitions(check_recovered PRIVATE GCS_HAVE_LZ4 GCS_HAVE_BZIP2 GCS_HAVE_SQLITE3)
target_link_libraries(check_recovered PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::Quick3D Qt6::Network Qt6::Concurrent SQLite::SQLite3 lz4 bz2)
